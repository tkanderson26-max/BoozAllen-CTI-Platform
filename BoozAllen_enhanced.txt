KEV_URL             = "https://www.cisa.gov/sites/default/files/feeds/known_exploited_vulnerabilities.json"
EPSS_URL            = "https://api.first.org/data/v1/epss?days=1"
RL_RECENT_URL       = "https://api.ransomware.live/v2/recentvictims"
RL_HISTORICAL_URL   = "https://api.ransomware.live/v2/victims/{year}/{month:02d}"

@st.cache_data(ttl=3600)
def fetch_kev():
    try:
        r = requests.get(KEV_URL, timeout=10)
        r.raise_for_status()
        return pd.DataFrame(r.json().get("vulnerabilities", []))
    except Exception:
        return pd.DataFrame()


@st.cache_data(ttl=3600)
def fetch_epss_snapshot():
    try:
        r = requests.get(EPSS_URL, timeout=10)
        r.raise_for_status()
        return pd.DataFrame(r.json().get("data", []))
    except Exception:
        return pd.DataFrame()


@st.cache_data(ttl=3600)
def fetch_ransomware_live(months_back: int = 12) -> tuple[pd.DataFrame, bool]:
    """
    ENHANCEMENT — Fetch live data from Ransomware.live API.

    Attempts:
      1. Recent victims endpoint  (last ~100 records)
      2. Monthly historical endpoints for `months_back` months

    Returns (DataFrame, is_live: bool).
    Falls back to synthetic data if API is unreachable, and sets is_live=False.
    """
    frames = []
    try:
        # 1 — Recent victims
        r = requests.get(RL_RECENT_URL, timeout=12)
        r.raise_for_status()
        recent = pd.DataFrame(r.json())
        if not recent.empty:
            frames.append(recent)

        # 2 — Monthly history
        today = datetime.today()
        for offset in range(months_back):
            yr  = (today.month - 1 - offset) // 12 + today.year if (today.month - 1 - offset) >= 0 else today.year - 1
            mo  = (today.month - 1 - offset) % 12 + 1
            yr  = today.year - (offset // 12) - (1 if (today.month - 1 - offset % 12) < 0 else 0)
            mo  = ((today.month - 1 - offset) % 12) + 1
            url = RL_HISTORICAL_URL.format(year=yr, month=mo)
            try:
                rh = requests.get(url, timeout=10)
                if rh.status_code == 200 and rh.text.strip() not in ("", "[]", "null"):
                    frames.append(pd.DataFrame(rh.json()))
            except Exception:
                pass
            time.sleep(0.3)   # respect undocumented rate limit

        if frames:
            df = pd.concat(frames, ignore_index=True).drop_duplicates()
            if "discovered" in df.columns:
                df["discovered"] = pd.to_datetime(df["discovered"], errors="coerce")
            # Normalise column names to match synthetic schema
            rename_map = {"title": "victim", "post_url": "url"}
            df.rename(columns={k: v for k, v in rename_map.items() if k in df.columns}, inplace=True)
            return df, True

    except Exception:
        pass

    # ── Fallback: synthetic data that mirrors live API schema ──────────────
    return _synthetic_ransomware(seed=2026, n=500), False


def _synthetic_ransomware(seed: int = 2026, n: int = 500) -> pd.DataFrame:
    """Reproducible synthetic dataset mirroring Ransomware.live API field schema."""
    rng = np.random.default_rng(seed)
    groups   = ["LockBit","ALPHV/BlackCat","Black Basta","Cl0p","Play","8Base",
                "Akira","INC Ransom","RansomHub","Medusa","Hunters International","NoEscape"]
    sectors  = ["Aerospace & Defense","Government / Public Sector","Defense Contractor",
                "DIB Supply Chain","Critical Infrastructure","Healthcare","Finance","Technology"]
    countries = ["United States","United Kingdom","Germany","Australia","Canada","France","Japan","Italy"]
    dates = pd.date_range("2022-06-01", "2026-03-31", periods=n)
    return pd.DataFrame({
        "group":      rng.choice(groups, n),
        "activity":   rng.choice(sectors, n),
        "country":    rng.choice(countries, n),
        "discovered": dates[rng.integers(0, n, n)],
        "victim":     [f"DIB-Org-{rng.integers(1000, 9999)}" for _ in range(n)],
    })

_years = list(range(2019, 2027))
_rows  = []
for _cat in threat_categories:
    _base  = _rng.integers(40, 160)
    _trend = np.linspace(0.75, 1.65, len(_years))
    _noise = _rng.normal(0, 7, len(_years))
    _vals  = np.maximum(0, _base * _trend + _noise).round(0)
    for _y, _v in zip(_years, _vals):
        _rows.append({
            "Year": _y, "Category": _cat,
            "Global": int(_v),
            "Local":  int(_v * _rng.uniform(0.35, 0.70)),
        })

    def render_diamond(title, adversary, capability, infra, victim):
        coords = {"Adversary": (0, 1.4), "Capability": (1.4, 0), "Infrastructure": (-1.4, 0), "Victim": (0, -1.4)}
        labels = {"Adversary": adversary, "Capability": capability, "Infrastructure": infra, "Victim": victim}
        edges  = [("Adversary","Capability"),("Adversary","Infrastructure"),("Capability","Victim"),("Infrastructure","Victim")]
        colors = {"Adversary": "#CC0000", "Capability": "#002147", "Infrastructure": "#4A90D9", "Victim": "#7B68EE"}
        fig = go.Figure()
        for a, b in edges:
            fig.add_trace(go.Scatter(
                x=[coords[a][0], coords[b][0]], y=[coords[a][1], coords[b][1]],
                mode="lines", line=dict(color="#C0C5C8", width=2), showlegend=False,
            ))
        for node, (x, y) in coords.items():
            fig.add_annotation(
                x=x, y=y, text=f"<b>{node}</b><br>{labels[node]}",
                showarrow=False, font=dict(color="white", size=9.5),
                bgcolor=colors[node], bordercolor="white",
                borderwidth=1, borderpad=7, opacity=0.95,
            )
        fig.update_layout(
            title=title, height=400, showlegend=False,
            xaxis=dict(visible=False, range=[-2.3, 2.3]),
            yaxis=dict(visible=False, range=[-2.3, 2.3]),
            paper_bgcolor="rgba(0,0,0,0)", plot_bgcolor="rgba(0,0,0,0)",
            margin=dict(l=20, r=20, t=50, b=20),
        )
        return fig

### Why This Approach Was Selected
A simple frequency count of KEV entries per vendor treats a CVE from 3 years ago the same as one added last week. This analytic layers **three complementary methods**:

1. **Exponential time decay** — `score = Σ exp(-λ × days_ago) × rw_weight` — weights recent CVEs exponentially higher
2. **Z-score normalization** — flags vendors statistically ≥ 2σ above the population mean (classical statistical inference)
3. **Isolation Forest (ML)** *(Enhancement)* — unsupervised anomaly detection on the multivariate feature space `[decay_score, kev_count, rw_cves]`. Detects outliers that Z-score alone may miss because they are anomalous on *multiple dimensions simultaneously*, not just one.

**Data Source:** CISA KEV (live) · **Tools:** pandas · numpy · scipy · scikit-learn · plotly
""")

    kev_scored = fetch_kev()
    if not kev_scored.empty and "vendorProject" in kev_scored.columns:
        kev_scored["dateAdded"] = pd.to_datetime(kev_scored["dateAdded"], errors="coerce")
        today = pd.Timestamp.today()
        kev_scored["days_ago"] = (today - kev_scored["dateAdded"]).dt.days.clip(lower=0).fillna(999)
        rw_col = "knownRansomwareCampaignUse" if "knownRansomwareCampaignUse" in kev_scored.columns else None
        kev_scored["rw_weight"] = 1.0
        if rw_col:

        st.markdown("#### 🎛️ Analytic Controls")
        c1, c2, c3 = st.columns(3)
        with c1:
            lambda_val = st.slider("Decay rate λ", 0.001, 0.02, 0.005, 0.001, format="%.3f")
        with c2:
            top_n = st.slider("Top N vendors", 5, 30, 15, 5)
        with c3:
            min_kev = st.slider("Min KEV entries", 1, 15, 2, 1)

        # ── Step 1: Exponential decay ──────────────────────────────────────
        kev_scored["decay_contrib"] = np.exp(-lambda_val * kev_scored["days_ago"]) * kev_scored["rw_weight"]
        vendor_decay = kev_scored.groupby("vendorProject")["decay_contrib"].sum().rename("decay_score")

                        if rw_col else pd.Series(dtype=int, name="rw_cves"))

        vendor_df = pd.concat([vendor_decay, vendor_count, rw_counts], axis=1).fillna(0)
        vendor_df = vendor_df[vendor_df["kev_count"] >= min_kev]

        # ── Step 2: Z-score ────────────────────────────────────────────────
        mu    = vendor_df["decay_score"].mean()
        sigma = vendor_df["decay_score"].std() + 1e-9
        vendor_df["z_score"]    = ((vendor_df["decay_score"] - mu) / sigma).round(2)
        vendor_df["z_anomaly"]  = vendor_df["z_score"] >= 2.0
        vendor_df["percentile"] = vendor_df["decay_score"].rank(pct=True).mul(100).round(1)

        features = vendor_df[["decay_score","kev_count","rw_cves"]].values

            vendor_df["dual_anomaly"]    = vendor_df["z_anomaly"] & vendor_df["if_anomaly"]
        else:

    st.markdown("""
### Why This Approach Was Selected
1. **Chi-square test of independence:** Tests whether the observed sector targeting distribution for each group departs significantly from random. A significant result (p < 0.05) means the group is *deliberately* choosing sectors.
2. **Linear regression trend inference:** Fits a linear model to each group's monthly victim count to estimate whether targeting is accelerating (positive slope), flat, or declining.

**Data Source:** Ransomware.live (live fetch with synthetic fallback) · **Tools:** scipy · plotly
""")

    with st.spinner("Fetching Ransomware.live data…"):
        df_rl_a2, rl_live_a2 = fetch_ransomware_live(months_back=12)

    st.info(f"{'🟢 Live Ransomware.live data' if rl_live_a2 else '🟡 Synthetic fallback — mirrors live API schema'}")

    # Ensure discovered is datetime
    df_rl_a2["discovered"] = pd.to_datetime(df_rl_a2.get("discovered"), errors="coerce")

    # ── Controls (SINGLE block — duplicate removed vs. original file) ──────
    st.markdown("#### 🎛️ Analytic Controls")
    ac1, ac2, ac3 = st.columns(3)
    with ac1:
        sectors_available = sorted(df_rl_a2["activity"].dropna().unique()) if "activity" in df_rl_a2.columns else ["Aerospace & Defense"]
        focus_sector = st.selectbox("Focus Sector", options=sectors_available, index=0, key="a2_sector")
    with ac2:
        top_groups_n = st.slider("Top N groups to profile", 3, 12, 6, 1, key="a2_top")
    with ac3:
        alpha = st.selectbox("Significance level (α)", [0.01, 0.05, 0.10], index=1, key="a2_alpha")

    # ── Chi-square ─────────────────────────────────────────────────────────
    contingency = pd.crosstab(df_rl_a2["group"], df_rl_a2["activity"]) if "group" in df_rl_a2.columns else pd.DataFrame()

    chi_results = []
    if not contingency.empty:
        for grp in contingency.index:
            row_obs   = contingency.loc[grp].values
            row_total = row_obs.sum()
            col_props = contingency.sum(axis=0) / contingency.sum().sum()
            expected  = (col_props * row_total).values
            col_idx   = list(contingency.columns).index(focus_sector) if focus_sector in contingency.columns else 0
            chi2_stat = ((row_obs[col_idx] - expected[col_idx])**2 / (expected[col_idx] + 1e-9))
            sector_pct = (row_obs[col_idx] / row_total * 100) if row_total > 0 else 0
            chi_results.append({
                "Group": grp,
                "Total Victims": int(row_total),
                f"{focus_sector} Victims": int(row_obs[col_idx]),
                "Expected (Random)": round(float(expected[col_idx]), 1),
                "χ² Contribution": round(float(chi2_stat), 2),
                "Sector %": round(float(sector_pct), 1),
                "Intentional Targeting": "✅ Yes" if chi2_stat > 3.84 else "❌ No",
            })

        try:
            chi2_full, p_full, dof_full, _ = chi2_contingency(contingency)
            overall_sig = p_full < alpha
        except Exception:
            chi2_full, p_full, dof_full, overall_sig = 0, 1, 0, False

    chi_df = pd.DataFrame(chi_results).sort_values("χ² Contribution", ascending=False) if chi_results else pd.DataFrame()

    # ── Linear trend ───────────────────────────────────────────────────────
    sector_df = df_rl_a2[df_rl_a2.get("activity","") == focus_sector].copy() if "activity" in df_rl_a2.columns else pd.DataFrame()
    if not sector_df.empty:
        sector_df["month_num"] = (sector_df["discovered"].dt.year - 2022) * 12 + sector_df["discovered"].dt.month
        sector_df["month_str"] = sector_df["discovered"].dt.to_period("M").astype(str)

    top_grps = chi_df.head(top_groups_n)["Group"].tolist() if not chi_df.empty else []
    trend_results = []
    for grp in top_grps:
        grp_monthly = (sector_df[sector_df["group"] == grp].groupby("month_num").size().reset_index(name="count")) if not sector_df.empty else pd.DataFrame()
        if len(grp_monthly) >= 3:
            slope, intercept, r_val, p_val, _ = linregress(grp_monthly["month_num"], grp_monthly["count"])

                "Group": grp,

                "p-value": round(p_val, 4),
                "Trend": "📈 Accelerating" if slope > 0.1 and p_val < alpha else
                         "📉 Declining"    if slope < -0.1 and p_val < alpha else
                         "➡️ Flat / No sig. trend",
            })
        else:
            trend_results.append({"Group": grp, "Slope (victims/month)": 0, "R²": 0, "p-value": 1.0, "Trend": "⚠️ Insufficient data"})

    if not chi_df.empty:

        with ch1:

def _build_triage_df():
    kev = fetch_kev()
    rows = []
    if not kev.empty:
        kev["dateAdded"] = pd.to_datetime(kev.get("dateAdded",""), errors="coerce")
        today = pd.Timestamp.today()
        for _, r in kev.head(80).iterrows():
            days = (today - r.get("dateAdded", today)).days if pd.notna(r.get("dateAdded")) else 999
            rw   = str(r.get("knownRansomwareCampaignUse","Unknown")).strip()
            sev  = "Critical" if (days <= 7 or rw.lower() == "known") else "High" if days <= 30 else "Medium" if days <= 90 else "Low"
            action_map = {"Critical":"Patch within 24 hrs — confirm on asset inventory, isolate if unpatched","High":"Patch within 72 hrs — add to sprint backlog, notify system owner","Medium":"Patch within 30 days — schedule in next maintenance window","Low":"Monitor — include in quarterly patch cycle"}
            rows.append({"CVE ID":r.get("cveID","N/A"),"Vendor":r.get("vendorProject","N/A"),"Product":r.get("product","N/A"),"Date Added":r.get("dateAdded","").strftime("%Y-%m-%d") if pd.notna(r.get("dateAdded")) else "N/A","Days Since Added":int(days),"Ransomware Use":rw,"Severity":sev,"Recommended Action":action_map[sev],"CMMC Control":"SI.1.210 / CM.2.061"})
    return pd.DataFrame(rows)
